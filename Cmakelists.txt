cmake_minimum_required(VERSION 3.20)
project(PlaylabsGL)

set(CMAKE_CXX_STANDARD 17)

# ---------------------------------------------------------------
# Source files shared by both the DLL and any consumer
# ---------------------------------------------------------------
set(ENGINE_SOURCES
    AABB.cpp
    Atlas.cpp
    Camera.cpp
    Input.cpp
    Sound.cpp
    Sprite.cpp
    TimelineAnimator.cpp
    Window.cpp
    PlaylabsGL.cpp
)

set(ENGINE_HEADERS
    AABB.h
    Atlas.h
    Camera.h
    Image.h
    Input.h
    Sound.h
    Sprite.h
    TimelineAnimator.h
    Timer.h
    Vec2.h
    Window.h
    PlaylabsGL.h
    json.hpp          # nlohmann/json — drop the single header here
)

# ---------------------------------------------------------------
# PlaylabsGL.dll
# ---------------------------------------------------------------
add_library(PlaylabsGL SHARED ${ENGINE_SOURCES} ${ENGINE_HEADERS} PlaylabsGL.def)

target_compile_definitions(PlaylabsGL PRIVATE PLAYLABSGL_EXPORTS)

target_include_directories(PlaylabsGL PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

target_link_libraries(PlaylabsGL
    opengl32
    winmm
    gdi32
)

# ---------------------------------------------------------------
# Game executable (replace main.cpp with your actual entry point)
# ---------------------------------------------------------------
add_executable(Game main.cpp)

target_link_libraries(Game PlaylabsGL)

# Copy DLL next to the executable after build
add_custom_command(TARGET Game POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:PlaylabsGL>
        $<TARGET_FILE_DIR:Game>
)
